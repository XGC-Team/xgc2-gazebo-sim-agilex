cmake_minimum_required(VERSION 3.16)
project(native_chassis_models LANGUAGES CXX)
find_package(XgcXrpc CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(Gazebo REQUIRED IMPORTED_TARGET gazebo)
get_target_property(options PkgConfig::Gazebo INTERFACE_COMPILE_OPTIONS)
list(FILTER options EXCLUDE REGEX "^-std=")
set_property(TARGET PkgConfig::Gazebo PROPERTY INTERFACE_COMPILE_OPTIONS "${options}")
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs sensor_msgs std_msgs tf2 tf2_ros)
foreach(value SCENE_INCLUDE SIMULATION_LIBRARY MATH_INCLUDE MECANUM_SOURCE)
  if(NOT ${value})
    message(FATAL_ERROR "${value} explicit native dependency is required")
  endif()
endforeach()
add_library(simulation_service SHARED IMPORTED)
set_target_properties(simulation_service PROPERTIES IMPORTED_LOCATION "${SIMULATION_LIBRARY}")
set(scout "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(scout_gazebo SHARED "${scout}/src/scout_skid_steer.cpp" "${scout}/src/scout_skid_steer_model_plugin.cpp")
add_library(scout_unicycle_plugin SHARED "${scout}/src/scout_unicycle_plugin.cpp")
add_library(scout_implicit_wheel_plugin SHARED "${scout}/src/scout_implicit_wheel_plugin.cpp")
add_library(gazebo_sim_mecanum_contract SHARED "${MECANUM_SOURCE}/src/mecanum_contract_plugin.cpp")
foreach(target scout_gazebo scout_unicycle_plugin scout_implicit_wheel_plugin gazebo_sim_mecanum_contract)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE "${scout}/include" "${SCENE_INCLUDE}" "${MATH_INCLUDE}")
  target_include_directories(${target} SYSTEM PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(${target} PRIVATE simulation_service XgcXrpc::http PkgConfig::Gazebo ${catkin_LIBRARIES})
endforeach()
pkg_check_modules(JsonCpp REQUIRED IMPORTED_TARGET jsoncpp)
add_library(xgc2_dispatcher_fixture SHARED dispatcher_fixture.cpp)
target_compile_features(xgc2_dispatcher_fixture PRIVATE cxx_std_20)
target_include_directories(xgc2_dispatcher_fixture PRIVATE "${SCENE_INCLUDE}")
target_include_directories(xgc2_dispatcher_fixture SYSTEM PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(xgc2_dispatcher_fixture PRIVATE simulation_service PkgConfig::Gazebo PkgConfig::JsonCpp ${catkin_LIBRARIES})
